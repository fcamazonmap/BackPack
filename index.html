function doGet(e) { var sheet = getDataSheet(); var rows = sheet.getDataRange().getValues(); for (var i = 0; i < rows.length; i++) { if (rows[i][0] === e.parameter.key) { return out({value: rows[i][1]}); } } return out({value: null}); } function doPost(e) { var body = JSON.parse(e.postData.contents); var sheet = getDataSheet(); var rows = sheet.getDataRange().getValues(); for (var i = 0; i < rows.length; i++) { if (rows[i][0] === body.key) { sheet.getRange(i + 1, 2).setValue(body.value); return out({ok: true}); } } sheet.appendRow([body.key, body.value]); return out({ok: true}); } function getDataSheet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); return ss.getSheetByName('data') || ss.insertSheet('data'); } function out(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }
